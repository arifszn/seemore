export const RELEASES = 'https://github.com/arifszn/seemore/releases/latest';

{/* Asset names carry the version, so links to files come from the latest release, read in
    the browser. Until it answers, or without JavaScript, they go to the release page. */}
export function useLatest() {
  const [release, setRelease] = useState(null);
  useEffect(() => {
    fetch('https://api.github.com/repos/arifszn/seemore/releases/latest')
      .then((res) => (res.ok ? res.json() : null))
      .then((json) => {
        if (json && typeof json.tag_name === 'string' && json.tag_name.startsWith('desktop-v')) setRelease(json);
      })
      .catch(() => {});
  }, []);
  return release;
}

export function assetUrl(release, pattern) {
  const asset = release?.assets?.find((a) => pattern.test(a.name));
  return asset ? asset.browser_download_url : RELEASES;
}

export const FILES = {
  macArm: /-mac-arm64\.dmg$/,
  macIntel: /-mac-x64\.dmg$/,
  windows: /^seemore-Setup-.*\.exe$/,
  deb: /_amd64\.deb$/,
  appImage: /\.AppImage$/,
};

export function detectPlatform() {
  const ua = navigator.userAgent;
  if (/Windows/i.test(ua)) return 'windows';
  if (/Mac OS X|Macintosh/i.test(ua)) return 'macos';
  if (/Linux/i.test(ua) && !/Android/i.test(ua)) return 'linux';
  return null;
}

{/* Apple Silicon or Intel, or null when the browser does not say. Chromium answers through
    client hints. Chrome and Firefox name the GPU in WebGL ("Apple M2", "Intel Iris"); Safari
    says "Apple GPU" on both, so it stays null there. */}
export async function detectMacArch() {
  try {
    const { architecture } = await navigator.userAgentData.getHighEntropyValues(['architecture']);
    if (architecture === 'arm') return 'arm';
    if (architecture === 'x86') return 'intel';
  } catch {}
  try {
    const gl = document.createElement('canvas').getContext('webgl');
    const info = gl.getExtension('WEBGL_debug_renderer_info');
    const renderer = gl.getParameter(info.UNMASKED_RENDERER_WEBGL);
    if (/Apple M\d/.test(renderer)) return 'arm';
    if (/Intel|AMD|Radeon|NVIDIA/i.test(renderer)) return 'intel';
  } catch {}
  return null;
}

export function PlatformIcon({ id }) {
  return (
    <svg className="seemore-download-icon" viewBox="0 0 24 24" aria-hidden="true">
      <path d={ICONS[id]} fill="currentColor" />
    </svg>

{/* One button per platform, each to its section below. The visitor's own platform becomes
    the primary button, which downloads its file directly where the file is known. On a Mac
    whose chip the browser does not reveal, it goes to the section instead. */}
export function PlatformButtons() {
  const release = useLatest();
  const [platform, setPlatform] = useState(null);
  const [macArch, setMacArch] = useState(null);
  useEffect(() => {
    const detected = detectPlatform();
    setPlatform(detected);
    if (detected === 'macos') detectMacArch().then(setMacArch);
  }, []);
  const macFile = macArch === 'arm' ? FILES.macArm : macArch === 'intel' ? FILES.macIntel : null;
  const buttons = [
    { id: 'macos', label: 'macOS', file: macFile, direct: 'Download for Mac' },
    { id: 'windows', label: 'Windows', file: FILES.windows, direct: 'Download for Windows' },
    // Linux has two files (.deb and AppImage), so its button goes to the section.
    { id: 'linux', label: 'Linux', file: null, direct: 'Download for Linux' },
  ];
  const ordered = [...buttons].sort((a, b) => (a.id === platform ? -1 : b.id === platform ? 1 : 0));
  const version = release ? release.tag_name.replace('desktop-v', '') : null;
  const other = macArch === 'intel'
    ? { label: 'Apple Silicon', file: FILES.macArm }
    : { label: 'Intel', file: FILES.macIntel };
  return (
    <div className="seemore-download">
      <div className="seemore-platforms">
        {ordered.map((b) =>
          b.id === platform ? (
            <a key={b.id} className="seemore-platform seemore-platform-primary" href={b.file ? assetUrl(release, b.file) : `#${b.id}`}>
              <PlatformIcon id={b.id} />
              {b.direct}
            </a>
          ) : (
            <a key={b.id} className="seemore-platform" href={`#${b.id}`}>
              <PlatformIcon id={b.id} />
              {b.label}
            </a>
          ),
        )}
      </div>
      <p className="seemore-download-note">
        {version ? `Version ${version}. ` : ''}Free and open source.
        {platform === 'macos' && macArch && (
          <>
            {' '}For {macArch === 'arm' ? 'Apple Silicon' : 'Intel'}. Not your Mac?{' '}
            <a href={assetUrl(release, other.file)}>Download for {other.label}</a>.
          </>
        )}
      </p>
    </div>
  );
}

export function formatSize(bytes) {
  return `${Math.round(bytes / 1e6)} MB`;
}

{/* One file to download: the platform's logo, a name, the file type and, once the release
    has answered, its size. */}
export function FileCard({ file, platform, title, type }) {
  const release = useLatest();
  const asset = release?.assets?.find((a) => file.test(a.name));
  return (
    <a className="seemore-file" href={asset ? asset.browser_download_url : RELEASES}>
      <span className="seemore-file-logo">
        <PlatformIcon id={platform} />
      </span>
      <span className="seemore-file-text">
        <span className="seemore-file-title">{title}</span>
        <span className="seemore-file-meta">{asset ? `${type} · ${formatSize(asset.size)}` : type}</span>
      </span>
      <svg className="seemore-file-arrow" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
        <path d="M12 4v12M6 11l6 6 6-6M5 20h14" />
      </svg>
    </a>
  );
}

# Desktop app

Open a folder or a `.md` file as a site in its own window. An integrated terminal opens in the site's folder: run an AI agent, git or a script there, and the pages update as the files change. Show or hide it with **View > Terminal** (<kbd>Ctrl</kbd>+<kbd>`</kbd>).
