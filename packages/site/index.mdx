{/* The promo film behind a cover: a poster for each theme and a play button. The video's own
    controls appear once it plays. */}
export function HeroVideo() {
  const video = useRef(null);
  const [started, setStarted] = useState(false);
  const play = () => {
    setStarted(true);
    video.current?.play();
  };
  return (
    <div className="seemore-shot seemore-shot--video seemore-hero-video">
      <video ref={video} src={promo} controls={started} muted playsInline preload="none"></video>
      {!started && (
        <button type="button" className="seemore-video-cover" aria-label="Play the seemore video" onClick={play}>
          <span className="seemore-poster seemore-poster-light" style={{ backgroundImage: `url(${poster})` }}></span>
          <span className="seemore-poster seemore-poster-dark" style={{ backgroundImage: `url(${posterDark})` }}></span>
          <span className="seemore-video-play">
            <svg viewBox="0 0 10 12" aria-hidden="true"><path d="M0 0L10 6L0 12z" /></svg>
          </span>
        </button>
      )}
    </div>
  );
}
